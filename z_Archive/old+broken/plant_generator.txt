const CROP_MODID = 'cslcrops'
const CROP_MODID_ = CROP_MODID + ':'

var tweaks = global.cslcrops_tweaks
var crops = tweaks.crops

StartupEvents.registry( 'block', event => {

    // Create Crops and Data
    global.forEachIn( crops, ( data, id ) => {
        const [ modid, name ] = id.split(':')
        var originalBlock = Block.getBlock( id )
        var params = {
            'itemModel': modid + ':item/' + data.seed.split(':')[1],
            'states': global.blockState.getOptions( originalBlock.blockStates[0] ),
            'blockstateJson': global.readJsonFileFromMod( 'assets', modid, 'blockstates/' + name + '.json' )
        }
        const { itemModel } = params 

        var cropId = CROP_MODID_ + name
        data.copycat = cropId
        buildDyingCrop(
            params,
            event.create( cropId, 'crop' )
                .bonemeal( context => { 
                    return copycatBonemeal( context, originalBlock )
                })
                .growTick( context => {
                    copycatCropRT( context, originalBlock )
                    return 0.0
                })
                .randomTick( context => {
                    return
                })
                .notSolid()
                .cropSoundType()
                .transparent( true )
                .noDrops()
                .item( item => {
                    item
                        .modelJson({
                            "parent": itemModel
                        })
                        .name( context => {
                            return id
                        })
                })
                .tag( 'cslcrops:crops' )
        ) // End function
    })
})

// Build Crops
function buildDyingCrop( params, blockBuilder ) {
    const { states, blockstateJson } = params

    addSeasonTags( params, blockBuilder ) // TODO
        
    assignStates( blockBuilder, states )
    setLayers( blockBuilder, states )
    blockBuilder.blockstateJson = blockstateJson

    return blockBuilder
}

    } catch ( error ) {
        console.error( error )
        global.say( 'Unable to bonemeal block' )
    }
    
    return originalIncrease
}

function assignStates( blockBuilder, states ) {
    global.forEachIn( states, ( state, name ) => {
        if ( name == 'age' ) {
            blockBuilder.property( BlockProperties[ 'AGE_' + ( state.options.length - 1 ) ] )
        } else {
            blockBuilder.property( BlockProperties[ name.toUpperCase() ] )
        } 
    })
}
function hasMatchingStates( state, testState ) {
    global.forEachIn( state, ( value, name ) => {
        if ( testState[ name ] == undefined || value != testState[ name ].value ) {
            return false
        }
    })
    return true
}
function setLayers( blockBuilder, states ) {
    var options = states.age.options
    blockBuilder.age( options.length - 1, shape => {
        for ( var age of options ) {
            var height = 2.0 + ( Math.floor( age / ( options.length - 1 ) )*7.0 )
            shape.shape( age, 0.0, 0.0, 0.0, 16.0, height, 16.0 )
        }
    } )
}
