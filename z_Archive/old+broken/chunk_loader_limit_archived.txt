    BlockEvents.placed( loader, event => {
        const { block, player } = event

        if ( player == undefined ) { // cancel if not player
            event.cancel()
        }

        registerLoader( block, player )
    })

    BlockEvents.broken( loader, event => {
        const { block, player } = event

        blockData = chunkLoaders.loaders[ posString( block ) ]
        if ( blockData.active ) {
            loaderDisabled( block )
        }

        unregisterLoader( block )
    })
}

    const { player } = event

    if ( playerData != undefined ) {
        playerData.lastOnline = Date.now().valueOf()
    }
})
PlayerEvents.loggedIn( event => {
    const { player } = event
    var playerData = chunkLoaders.players[ player.uuid.toString() ]
    if ( playerData != undefined ) {
        playerData.lastOnline = Date.now().valueOf()
    }
})

// Managing Functions
function registerPlayer( player ) {
    chunkLoaders.players[ player.uuid.toString() ] = {
        loaders: [],
        loadedChunks: 0,
        lastOnline: Date.now().valueOf(),
        username: player.username
    }
}
function registerLoader( block, player ) {
    var playerData = chunkLoaders.players[ player.uuid.toString() ]
    if ( playerData == undefined ) {
        registerPlayer( player )
    }

    playerData.loaders.push( posString( block ) )

    chunkLoaders.loaders[ posString( block ) ] = {
        owner: player.uuid.toString(),
        active: false,
        type: block.id,
        range: 0,
        position: {
            x: block.x,
            y: block.y,
            z: block.z
        },
        overlaps: [],
        effectiveAreaCache: 1,
        cacheValid: false
    }

    registerOverlaps( block )
}
function unregisterLoader( block ) {
    var blockData = chunkLoaders.loaders[ posString( block ) ]
    var playerData = chunkLoaders.players[ blockData.owner ]

    // Remove Overlap Refs
    for ( var loader of blockData.overlaps ) {
        global.removeElement( chunkLoaders.loaders[ loader ], posString( block ) )
    }

    global.removeElement( playerData.loaders, posString( block ) )
    delete chunkLoaders.loaders[ posString( block ) ]
}
function disableLoader( block ) {
    var blockData = chunkLoaders.loaders[ posString( block ) ]
    if ( blockData == undefined ) {
        console.error( 'Attempted to reset block at ' + posString( block ) + ': Not registered.' )
        return
    }
    if ( !blockData.active ) {
        console.error( 'Attempting to disable chunk loader ' + posString( block ) + ': Not Active.' )
        return
    }

    const { x, y, z } = block
    var loc = global.new.point( x, y, z )

    // Break create network
    global.run( 'data modify block ' + loc.toCommandString() + ' Network.Size set value 0' )

    notifyPlayer( chunkLoaders.players[ block.owner ].username )
    loaderDisabled( block )
}
function registerOverlaps( block ) {
    // Get Data Paths
    var blockData = chunkLoaders.loaders[ posString( block ) ]
    var playerData = chunkLoaders.players[ blockData.owner ]

    // Determine chunk
    var aChunk = chunk( block.x, block.z )
    var thisLoc = global.new.point( aChunk.x, 0, aChunk.z )
    // Check all player loaders
    for ( var loader of playerData.loaders ) {
        // Get Data of other loader
        var secondBlockData = chunkLoaders.loaders[ loader ]
        const { x: xs, z: zs } = secondBlockData.position
        aChunk = chunk( xs, zs )

        // Determine Intersection
        var chunkDist = thisLoc.distanceCardinal( global.new.point( aChunk.x, 0, aChunk.z ) )
        var maxDist = maxRange( blockData.type ) + maxRange( secondBlockData.type )
        if ( maxDist >= chunkDist.x || maxDist >= chunkDist.z ) {
            blockData.overlaps.push( loader )
            secondBlockData.overlaps.push( posString( block ) )
            secondBlockData.cacheValid = false
        }
    }
}

// Updator Functions
function loaderEnabled( block ) {
    var blockData = chunkLoaders.loaders[ posString( block ) ]
    var playerData = chunkLoaders.players[ blockData.owner ]

    updateOverlaps( playerData )

    playerData.loadedChunks += effectiveArea( blockData )
    blockData.active = true
}
function loaderDisabled( block ) {
    blockData = chunkLoaders.loaders[ posString( block ) ]
    playerData = chunkLoaders.players[ blockData.owner ]

    updateOverlaps( playerData )

    playerData.loadedChunks -= effectiveArea( blockData )
    blockData.active = false
}

        return blockData.effectiveAreaCache

        var maxDist = blockData.range + overlapLoader.range

}
function updateOverlaps( playerData ) {
    for ( var loader of playerData.loaders ) {
        chunkLoaders.loaders[ loader ].cacheValid = false
    }
}
