        joined: {}
    }

    event.player.closeMenu()
} )

var playerJoined = global.curio_lantern.joined
PlayerEvents.loggedIn( event => {
    playerJoined[ event.player.username ] = true
    // updateLantern( event.player )
})

// Detect tampering
global.addListener(
    'top.theillusivec4.curios.api.event.CurioEquipEvent',
    'curio_lantern_add',
    ( event ) => {
        const { stack, slotContext, entity: player } = event

        Utils.runAsync( () => {
            if ( !slotContext.identifier().equals('lantern') || !event.entity.isPlayer() || playerJoined[player.username] == undefined ) {
                return
            }

            fillLantern( player, stack.item.id )
        })
    }
)
global.addListener(
    'top.theillusivec4.curios.api.event.CurioUnequipEvent',
    'curio_lantern_remove',
    ( event ) => {
        const { stack, slotContext, entity: player } = event

        Utils.runAsync( () => {
            if ( !slotContext.identifier().equals('lantern') || !event.entity.isPlayer() || playerJoined[player.username] == undefined ) {
                return
            }

            setTimeout(() => {
                if ( stack.item.id.equals('minecraft:air') ) {
                    clearLantern( player )
                }
            }, 50);
                
            
            if ( stack.item.id.equals('minecraft:air') ) {
                clearLantern( player )
            }
        })
    }
)

// Operation Handlers
function clearLantern( player ) {
    var nbt = player.nbt
    var accessories = nbt["fabric:attachments"]["accessories:inventory_holder"]
    if ( accessories.AccessoriesContainers == undefined ) {
        return
    }

    var items = accessories.AccessoriesContainers.belt.Items
    if ( items.size() > 0 ) {
        items[0] = {}
        player.setNbt(nbt)
    }
}
function fillLantern( player, item ) {
    var newItem = {Count:1,Slot:0,id:"minecraft:lantern"}
    if ( item.equals( "minecraft:soul_lantern" ) ) {
        newItem = {Count:1,Slot:0,id:"minecraft:soul_lantern"}
    }

    var nbt = player.nbt
    var accessories = nbt["fabric:attachments"]["accessories:inventory_holder"]
    if ( accessories.AccessoriesContainers == undefined ) {
        return
    }
    
    var items = accessories.AccessoriesContainers.belt.Items
    if ( items.size() == 0 ) {
        items.push(newItem)
        player.setNbt(nbt)
    }
}
function updateLantern ( player ) {
    var nbt = player.nbt
    var curios = nbt.ForgeCaps["curios:inventory"].Curios

    var slotItem
    for ( var curio of curios ) {
        if ( curio.Identifier.equals("lantern") ) {
            if ( !curio.StacksHandler.Stacks.Items == undefined ) {
                slotItem = curio.StacksHandler.Stacks.Items[0].id
            }
            break
        }
    }
    if ( slotItem != undefined ) {
        fillLantern( player, slotItem )
    } else {
        clearLantern( player )
    }
}
